# The following five lines of boilerplate have to be in your project's
# CMakeLists in this exact order for cmake to work correctly
cmake_minimum_required(VERSION 3.16)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(kestrel-interface)

# Keep firmware debug-friendly while optimizing LVGL's per-pixel drawing loops.
idf_component_get_property(lvgl_lib lvgl__lvgl COMPONENT_LIB)
target_compile_options(${lvgl_lib} PRIVATE -O2)

# Isolated renderer experiment; normal firmware retains the checked backend.
option(KEST_ROUNDED_FILL_PREVIEW "Try PPA interiors for software rounded fills" OFF)
option(KEST_ROUNDED_CORNERS_PREVIEW "Compare three PPA rectangles with software corners" OFF)
option(KEST_DRAW_PROFILE_PREVIEW "Time drawing paths in the isolated corner renderer" OFF)
option(KEST_ROUNDED_CACHE_PREVIEW "Use the pinned PPA driver's cache synchronization" OFF)
if(KEST_ROUNDED_CACHE_PREVIEW AND NOT (KEST_ROUNDED_FILL_PREVIEW AND KEST_ROUNDED_CORNERS_PREVIEW))
    message(FATAL_ERROR "Rounded cache preview requires both rounded preview options")
endif()
if(KEST_DRAW_PROFILE_PREVIEW AND NOT (KEST_ROUNDED_FILL_PREVIEW AND KEST_ROUNDED_CORNERS_PREVIEW))
    message(FATAL_ERROR "Drawing profile requires both rounded preview options")
endif()
if(KEST_ROUNDED_FILL_PREVIEW)
    if(KEST_ROUNDED_CACHE_PREVIEW)
        target_compile_definitions(${lvgl_lib} PRIVATE KEST_ROUNDED_DRIVER_CACHE_PREVIEW=1)
    endif()
    if(KEST_DRAW_PROFILE_PREVIEW)
        target_sources(${lvgl_lib} PRIVATE tools/draw_profile_preview.c)
        foreach(draw border label arc image box_shadow)
            target_link_libraries(${lvgl_lib} INTERFACE "-Wl,--wrap=lv_draw_sw_${draw}")
        endforeach()
    elseif(KEST_ROUNDED_CORNERS_PREVIEW)
        target_sources(${lvgl_lib} PRIVATE tools/rounded_corners_preview.c)
    else()
        target_sources(${lvgl_lib} PRIVATE tools/rounded_fill_preview.c)
    endif()
    target_link_libraries(${lvgl_lib} INTERFACE "-Wl,--wrap=lv_draw_sw_fill")
    idf_component_get_property(main_lib main COMPONENT_LIB)
    target_sources(${main_lib} PRIVATE tools/rounded_fill_check.c)
    target_link_libraries(${main_lib} PRIVATE idf::esp_driver_ppa)
    target_link_libraries(${main_lib} INTERFACE "-Wl,--wrap=ppa_do_fill")
    target_compile_definitions(${main_lib} PRIVATE KEST_ROUNDED_FILL_PREVIEW=1)
    if(KEST_DRAW_PROFILE_PREVIEW)
        target_compile_definitions(${main_lib} PRIVATE KEST_DRAW_PROFILE_PREVIEW=1)
    endif()
endif()

idf_build_set_property(COMPILE_OPTIONS "-Wno-error=format" APPEND)
